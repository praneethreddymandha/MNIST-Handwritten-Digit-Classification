# -------------------------
# Section 2: Data Loading & Exploration
# -------------------------
# Load MNIST dataset
(train_images, train_labels), (test_images, test_labels) = mnist.load_data()

# Display dataset summary
print("\nTraining data shape:")
print("Images:", train_images.shape)
print("Labels:", train_labels.shape)
print("\nTest data shape:")
print("Images:", test_images.shape)
print("Labels:", test_labels.shape)

# Visualize sample images
plt.figure(figsize=(10,5))
for i in range(12):
    plt.subplot(3, 4, i+1)
    plt.imshow(train_images[i], cmap='gray')
    plt.title(f"Label: {train_labels[i]}")
    plt.axis('off')
plt.tight_layout()
plt.show()

# -------------------------
# Section 3: Data Preprocessing
# -------------------------
# Normalize pixel values
train_images = train_images.astype('float32') / 255
test_images = test_images.astype('float32') / 255

# Reshape for CNN input
train_images = train_images.reshape((60000, 28, 28, 1))
test_images = test_images.reshape((10000, 28, 28, 1))

# Convert labels to categorical
train_labels = to_categorical(train_labels)
test_labels = to_categorical(test_labels)

# -------------------------
# Section 4: Model Building
# -------------------------
# Create CNN model
model = Sequential([
    Conv2D(32, (3,3), activation='relu', input_shape=(28,28,1)),
    MaxPooling2D((2,2)),
    Conv2D(64, (3,3), activation='relu'),
    MaxPooling2D((2,2)),
    Flatten(),
    Dense(128, activation='relu'),
    Dropout(0.5),
    Dense(10, activation='softmax')
])

# Compile the model
model.compile(optimizer='adam',
              loss='categorical_crossentropy',
              metrics=['accuracy'])

# Display model summary
model.summary()

# -------------------------
# Section 5: Model Training
# -------------------------
# Train the model

history = model.fit(train_images, train_labels,
                    epochs=10,
                    batch_size=64,
                    validation_data=(test_images, test_labels))

# -------------------------
# Section 6: Model Evaluation
# -------------------------
# Evaluate on test set
test_loss, test_acc = model.evaluate(test_images, test_labels)
print(f"\nTest Accuracy: {test_acc:.4f}")
print(f"Test Loss: {test_loss:.4f}")

# Plot training history

# --- Tkinter App with Input Display ---
class DigitDrawingApp:
    def __init__(self, master):
        self.master = master
        master.title("Draw a Digit!")

        self.CANVAS_SIZE = 300
        self.canvas = Canvas(master, width=self.CANVAS_SIZE, height=self.CANVAS_SIZE, bg='black')
        self.canvas.pack()

        self.predict_button = Button(master, text="Predict", command=self.predict_digit)
        self.predict_button.pack(pady=5)

        self.clear_button = Button(master, text="Clear", command=self.clear_canvas)
        self.clear_button.pack(pady=5)

        self.label = Label(master, text="Prediction: ", font=("Arial", 16))
        self.label.pack(pady=5)

        # Image preview (28x28 input to model)
        self.image_label = Label(master)
        self.image_label.pack(pady=10)

        # For drawing
        self.image = Image.new("L", (self.CANVAS_SIZE, self.CANVAS_SIZE), color=0)
        self.draw = ImageDraw.Draw(self.image)

        self.canvas.bind("<B1-Motion>", self.paint)

    def paint(self, event):
        x1, y1 = (event.x - 10), (event.y - 10)
        x2, y2 = (event.x + 10), (event.y + 10)
        self.canvas.create_oval(x1, y1, x2, y2, fill="white", outline="white", width=20)
        self.draw.ellipse((x1, y1, x2, y2), fill=255)

    def clear_canvas(self):
        self.canvas.delete("all")
        self.image = Image.new("L", (self.CANVAS_SIZE, self.CANVAS_SIZE), color=0)
        self.draw = ImageDraw.Draw(self.image)
        self.label.config(text="Prediction: ")
        self.image_label.config(image="")

    def predict_digit(self):
        # Resize for model input
        img = self.image.resize((28, 28)).convert('L')
        img_array = np.array(img).reshape(1, 28, 28, 1) / 255.0

        # Prediction
        prediction = model.predict(img_array)
        digit = np.argmax(prediction)
        confidence = np.max(prediction) * 100
        self.label.config(text=f"Prediction: {digit} ({confidence:.2f}% confidence)")

        # Show input image
        img_preview = img.resize((140, 140)).convert("L")  # Make it big enough to view
        img_preview = ImageTk.PhotoImage(img_preview)
        self.image_label.config(image=img_preview)
        self.image_label.image = img_preview  # Keep a reference!

        # Matplotlib-style plot in notebook
        plt.figure(figsize=(3, 3))
        plt.imshow(img_array[0].reshape(28, 28), cmap='gray')
        plt.title(f"Input to Model\nPrediction: {digit} ({confidence:.2f}%)")
        plt.axis('off')
        plt.tight_layout()
        plt.show()

# Launch the app
if __name__ == '__main__':
    root = Tk()
    app = DigitDrawingApp(root)
    root.mainloop()
